% =================================================================
% Molecule helper: places coordinates for a regular hexagonal
% pyridone-like molecule. Usage: \molcoords{<prefix>}{<Rring>}
% Generates: <pref>top, <pref>tr, <pref>br, <pref>bot, <pref>N, <pref>tl,
%            <pref>O (carbonyl), <pref>Nme (N-methyl C)
% =================================================================
\newcommand{\molcoords}[2]{%
  \coordinate (#1top) at ( 90:#2);
  \coordinate (#1tr)  at ( 30:#2);
  \coordinate (#1br)  at (-30:#2);
  \coordinate (#1bot) at (-90:#2);
  \coordinate (#1N)   at (210:#2);
  \coordinate (#1tl)  at (150:#2);
  \coordinate (#1O)   at ($ (#1top) + ( 90:#2) $);
  \coordinate (#1Nme) at ($ (#1N)   + (210:#2) $);
}

% =================================================================
\newcommand{\msgring}[3]{% #1=xshift, #2=yshift (box center), #3=unique prefix
\begin{scope}[shift={(#1,#2-0.25)}]
    \molcoords{#3}{0.55}
    % Ring + exocyclic bonds (lighter purple, behind atoms)
    \draw[thick, RoyalPurple!55] (#3top) -- (#3tr);
    \draw[thick, RoyalPurple!55] (#3tr)  -- (#3br);
    \draw[thick, RoyalPurple!55] (#3br)  -- (#3bot);
    \draw[thick, RoyalPurple!55] (#3bot) -- (#3N);
    \draw[thick, RoyalPurple!55] (#3N)   -- (#3tl);
    \draw[thick, RoyalPurple!55] (#3tl)  -- (#3top);
    \draw[thick, RoyalPurple!55] (#3top) -- (#3O);
    \draw[thick, RoyalPurple!55] (#3N)   -- (#3Nme);
    % Atoms (named so we can attach message arrows)
    \node[aP] (#3-top) at (#3top) {C};
    \node[aP] (#3-tr)  at (#3tr)  {C};
    \node[aP] (#3-br)  at (#3br)  {C};
    \node[aP] (#3-bot) at (#3bot) {C};
    \node[aP] (#3-N)   at (#3N)   {N};
    \node[aP] (#3-tl)  at (#3tl)  {C};
    \node[aP] (#3-O)   at (#3O)   {O};
    \node[aP] (#3-Nme) at (#3Nme) {C};
    % Bidirectional message-passing arrows along every bond
    \foreach \a/\b in {top/tr, tr/br, br/bot, bot/N, N/tl, tl/top, top/O, N/Nme} {
        \draw[msgar] (#3-\a) to[bend left=14] (#3-\b);
        \draw[msgar] (#3-\b) to[bend left=14] (#3-\a);
    }
\end{scope}}
