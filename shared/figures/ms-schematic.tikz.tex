\begin{tikzpicture}[
  fig,
  font=\sffamily\scriptsize,
  every node/.style={align=center},
  lbl/.style={draw, thick, fill=white, inner sep=2.5pt},
  smalllbl/.style={draw, thick, fill=white, inner sep=2pt},
  wire/.style={thick},
  beam/.style={very thick},
  arrow/.style={->, thick},
  hit/.style={fill=black, draw=none},
]

% ============================================================
% GEOMETRY : one straight horizontal flight tube (no bend)
% ============================================================
\def\h{0.35}        % tube half-width
\def\Xmag{3.9}      % magnet left edge
\def\Wmag{1.7}      % magnet width
\def\Xhit{5.15}     % where the off-mass beams reach the walls
\def\Xdet{6.8}      % detector mouth

% ---------- ion source / acceleration block ----------
\draw[arrow] (-2.1,0.0) -- (-0.35,0.0);
\node[anchor=east] at (-2.15,0.0) {vaporised\\sample};
\draw[wire] (-0.35,-0.35) rectangle (2.15,0.35);
\draw[wire] (0.15,-0.05) .. controls (0.35,0.25) and (0.55,0.25) .. (0.75,-0.05);
\draw[wire] (0.75,-0.05) .. controls (0.55,-0.35) and (0.35,-0.35) .. (0.15,-0.05);
\foreach \x in {1.10,1.35,1.60}{\draw[wire] (\x,-0.35) -- (\x,0.35);}

% ============================================================
% ELECTROMAGNET : two straight pole pieces, above and below the tube
% ============================================================
\filldraw[fill=Tan, draw=Tan!55!black, thick, rounded corners=2pt]
  (\Xmag, 0.50) rectangle ($(\Xmag,0.50)+(\Wmag,0.70)$);     % top pole
\filldraw[fill=Tan, draw=Tan!55!black, thick, rounded corners=2pt]
  (\Xmag,-0.50) rectangle ($(\Xmag,-0.50)+(\Wmag,-0.70)$);   % bottom pole

% ============================================================
% FLIGHT TUBE : straight horizontal walls
% ============================================================
\draw[wire] (2.15, \h) -- (\Xdet, \h);
\draw[wire] (2.15,-\h) -- (\Xdet,-\h);

% ============================================================
% ION STREAMS : correct one runs dead straight; others fan to the walls
% ============================================================
\draw[beam, green!55!black] (2.15,0) -- (\Xdet,0);                       % correct -> detector
\draw[beam, blue] (\Xmag,0) .. controls ($(\Xmag,0)+(0.5,0)$) and (\Xhit-0.30, 0.30) .. (\Xhit, \h)
   coordinate (hitH);                                                    % less deflected -> top wall
\draw[beam, red]  (\Xmag,0) .. controls ($(\Xmag,0)+(0.5,0)$) and (\Xhit-0.30,-0.30) .. (\Xhit,-\h)
   coordinate (hitL);                                                    % more deflected -> bottom wall
\fill[hit] (hitH) circle (1.6pt);
\fill[hit] (hitL) circle (1.6pt);

% ============================================================
% DETECTOR + ELECTRONICS
% ============================================================
\draw[wire] (\Xdet,-0.40) rectangle ($(\Xdet,-0.40)+(0.35,0.80)$);
\node[smalllbl, minimum width=1.5cm, minimum height=0.55cm]
  (amp) at ($(\Xdet,0)+(1.95,-0.15)$) {amplifier\\ \& other electronics};
\draw[arrow] ($(\Xdet,0)+(0.37,0)$) -- (amp.west);

% ============================================================
% LABELS
% ============================================================
\node[lbl] (ionlab) at (0.2,2.05) {IONISATION};
\draw[arrow] (ionlab.south) -- (0.45,0.38);
\node[lbl] (acclab) at (2.8,2.15) {ACCELERATION};
\draw[arrow] (acclab.south west) -- (1.70,0.40);
\node[lbl] (magn) at (5.6,2.05) {ELECTROMAGNET};
\draw[arrow] (magn.south) -- ($(\Xmag+\Wmag-0.5,1.20)$);
\node[lbl] (deflab) at (3.7,-2.05) {DEFLECTION};
\draw[arrow] (deflab.north) -- (4.55,-0.16);
\node[lbl] (detlab) at (6.55,-1.55) {DETECTION};
\draw[arrow] (detlab.north) -- ($(\Xdet,0)+(0.05,-0.32)$);
