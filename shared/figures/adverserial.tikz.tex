\begin{tikzpicture}[
  font=\small,
  box/.style={draw, rounded corners, align=left, inner sep=6pt, line width=0.6pt},
  head/.style={draw, rounded corners, align=center, inner sep=6pt, line width=0.8pt},
  line/.style={-Latex, line width=0.7pt},
  flowdash/.style={-Latex, dashed, line width=0.7pt},
  note/.style={align=left},
  lbl/.style={font=\small\bfseries},
  tight/.style={inner sep=4pt},
  coltitle/.style={font=\small\bfseries}
]

% --- INPUT ---------------------------------------------------------------
\node[head] (phase) {PHASE A: ADVERSARIAL};

\node[box, below=8pt of phase] (input) {%
\textbf{INPUT BATCH}\\
$\bullet$ Molecule Graph (\texttt{GeometricData})\\
$\bullet$ Fragment Graphs (\texttt{List[GeometricData]})\\
$\bullet$ Fragment Masses (Tensor)\\
$\bullet$ Real Spectrum (Tensor)\\
$\bullet$ SMILES strings
};

% --- ENCODING ------------------------------------------------------------
\node[box, below=10pt of input] (enc) {%
\textbf{ENCODING PHASE}\\
EncMol(mol\_graph) $\rightarrow z_m\;\;[B,d_{\text{latent}}]$\\
FragSetEnc(frag) $\rightarrow z_f\;\;[B,d_{\text{latent}}]$\\
$(z_m+z_f)/2 \rightarrow z\;\;[B,d_{\text{latent}}]$\\
heads($z$) $\rightarrow z_{\text{fwd}}\;\;[B,d_{\text{latent}}]$
};

% --- GENERATION ----------------------------------------------------------
\node[box, below=10pt of enc] (gen) {%
\textbf{GENERATION PHASE}\\
SpecGenerator($z_{\text{fwd}}$) $\rightarrow$ \texttt{fake\_spec}
};

% --- DISCRIMINATOR LOOP --------------------------------------------------
\node[box, below=10pt of gen, minimum width=0.88\linewidth] (disc) {%
\textbf{DISCRIMINATOR LOOP (5 steps)}\\[2pt]
\textbf{[A]} $D(\text{real}) \leftarrow$ spec\_disc(real\_spec)\\
\textbf{[B]} $D(\text{fake}) \leftarrow$ spec\_disc(fake\_spec.\texttt{detach()})\\
\hspace*{1.5em}\emph{(no gradient to SpecGenerator here)}\\[2pt]
$L_{\text{disc}}=\text{BCE}(D_{\text{real}},1)+\text{BCE}(D_{\text{fake}},0)$\\[2pt]
\texttt{discriminator.backward()}\\
\texttt{disc\_optimizer.step()}
};

% --- GENERATOR STEP ------------------------------------------------------
\node[box, below=10pt of disc, minimum width=0.88\linewidth] (gstep) {%
\textbf{GENERATOR STEP (1 update)}\\[4pt]
\textbf{Adversarial loss}\\
$D(\text{fake}) \leftarrow$ spec\_disc(fake\_spec) \;\;[\emph{no detach}]\\
$L_{\text{gen\_adv}}=\text{BCE}(D_{\text{fake}},1)$\\[3pt]
\textbf{Reconstruction loss (0.1$\times$)}\\
Apply parent-mass mask to \texttt{fake\_spec}\\
$L_{\cos}=\text{cosine}(\text{fake\_masked},\text{real\_spec})$\\
$L_{\text{wass}}=\text{wasserstein}(\text{fake\_masked},\text{real\_spec})$\\
$L_{\text{recon}}=L_{\cos}+L_{\text{wass}}$\\[3pt]
\textbf{Cycle consistency (0.1$\times$)}\\
$z_{\text{spec\_cycle}}=\text{enc\_spec}(\text{fake\_masked}.\texttt{detach()})$\\
$L_{\text{cycle}}=\text{cosine}(z_{\text{spec\_cycle}}, z_m)$\\[3pt]
\textbf{Combined loss}\\
$L_{\text{total}}=L_{\text{gen\_adv}}+0.1\,L_{\text{recon}}+0.1\,L_{\text{cycle}}$\\[3pt]
$L_{\text{total}}.\texttt{backward()}$\\
\texttt{gen\_optimizer.step()} \;[\textit{updates: enc\_mol, frag\_enc, spec\_gen, heads}]
};

% --- METRICS -------------------------------------------------------------
\node[box, below=10pt of gstep] (metrics) {%
\textbf{ACCUMULATE METRICS}\\
avg\_gen\_loss $+= L_{\text{gen}}\cdot$ batch\_size\\
avg\_disc\_loss $+= L_{\text{disc}}\cdot$ batch\_size\\
disc\_acc $= \dfrac{\mathbb{P}(D_{\text{real}}>0)+\mathbb{P}(D_{\text{fake}}<0)}{2}$ \;\;[\emph{should} $\rightarrow 0.5$]
};

% --- OUTPUT --------------------------------------------------------------
\node[box, below=10pt of metrics] (out) {%
\textbf{OUTPUT (per epoch)}\\
$\bullet$ G\_loss: average generator loss\\
$\bullet$ D\_loss: average discriminator loss\\
$\bullet$ D\_acc: discriminator accuracy ($\approx 0.5$ = good)
};

% --- ARROWS MAIN FLOW ----------------------------------------------------
\draw[line] (phase) -- (input);
\draw[line] (input) -- (enc);
\draw[line] (enc) -- (gen);
\draw[line] (gen) -- (disc);
\draw[line] (disc) -- (gstep);
\draw[line] (gstep) -- (metrics);
\draw[line] (metrics) -- (out);

% --- SIDE CONNECTIONS / LABELS ------------------------------------------
% Real spectrum goes to discriminator [A]
\coordinate (realTap) at ($(input.east)+(0.1,0)$);
\draw[line] (input.east) -- ++(1.0,0) |- node[pos=0.25, right, tight]{\textbf{real\_spec}} (disc.north east);

% Fake spectrum goes to discriminator [B] from generation
\draw[line] (gen.east) -- ++(1.0,0) |- node[pos=0.25, right, tight]{\textbf{fake\_spec.detach()}} ($(disc.east)+(0,0.4)$);

% Fake spectrum (no detach) used in generator adversarial loss
\draw[flowdash] (gen.west) -- ++(-1.0,0) |- node[pos=0.25, left, tight]{\textbf{fake\_spec (no detach)}} ($(gstep.west)+(0,2.0)$);

% Emphasize detach points
\node[note, right=10pt of disc] (dnote) {%
\textbf{Detach points}\\
\quad$\rightarrow$ Disc update: \texttt{fake\_spec.detach()}\\
\quad$\rightarrow$ Cycle: \texttt{fake\_masked.detach()}
};

\end{tikzpicture}
