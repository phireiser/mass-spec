\begin{tikzpicture}[
  font=\small,
  >=Latex,
  node distance=10mm and 14mm,
  box/.style={draw, rounded corners=2pt, align=left, inner sep=6pt, minimum width=44mm},
  bigbox/.style={draw, rounded corners=2pt, align=left, inner sep=8pt, minimum width=62mm},
  title/.style={font=\bfseries, align=center},
  arrow/.style={-Latex, line width=0.6pt},
  dashedarrow/.style={-Latex, dashed, line width=0.6pt},
  pill/.style={draw, rounded corners=10pt, inner sep=4pt, align=center},
  loss/.style={draw, rounded corners=2pt, inner sep=5pt, align=left, minimum width=62mm},
  group/.style={draw, rounded corners=2pt, inner sep=10pt}
]

% =========================
% TOP: TRAINING ARCHITECTURE
% =========================
\node[title] (archTitle) {Fragment-Aided Bidirectional Adversarial GAN Model Architecture};

\node[bigbox, below=10mm of archTitle] (input) {\textbf{INPUT DATA}\\
\(\bullet\) Molecule Graphs (M\O D)\\
\(\bullet\) Derivation Trees (Fwd \& Bwd)\\
\(\bullet\) Real Spectra (binned)};

% Encoders
\node[box, below left=14mm and 18mm of input] (encMol) {\textbf{MOLECULE ENCODER} (EncMol)\\
Input: Mol graph\\
GCN \(3 \rightarrow 64 \rightarrow d\)\\
Global mean pool\\
Output: \(z_{\text{mol}}\in\mathbb{R}^{B\times d}\)};

\node[box, below right=14mm and 18mm of input] (encSpec) {\textbf{SPECTRUM ENCODER} (EncSpec)\\
Input: binned spec\\
MLP \(n \rightarrow 512 \rightarrow d\)\\
+ normalization\\
Output: \(z_{\text{spec}}\in\mathbb{R}^{B\times d}\)};

% Discriminator + contrastive
\node[box, below=10mm of encSpec] (disc) {\textbf{DISCRIMINATOR}\\(SpecDiscriminator)\\
MLP classifier\\
Real vs fake spec\\
Output: logits \([B,1]\)};

\node[pill, right=10mm of disc] (infonce) {\textbf{Contrastive Loss}\\InfoNCE};

% Fragment encoder
\node[box, below=14mm of encMol] (fragEnc) {\textbf{FRAGMENT SET ENCODER}\\(FragSetEncoderWrapper)\\
Input: derivation trees\\
\(\bullet\) Fwd: Mol \(\rightarrow\) fragments\\
\(\bullet\) Bwd: fragments \(\rightarrow\) Mol\\
GCN + attention pooling\\
Output: \(z_{\text{frag}}\in\mathbb{R}^{B\times d}\)};

% Fusion
\node[pill, below=10mm of fragEnc, minimum width=44mm] (fusion) {\textbf{LATENT FUSION}\\
\(z=\dfrac{z_{\text{mol}}+z_{\text{frag}}}{2}\)};

% Task heads
\node[bigbox, below=10mm of fusion] (heads) {\textbf{TASK HEADS (Branching)} (TaskHeads)\\[1mm]
\textbf{Forward head (Mol \(\rightarrow\) Spec)}\\
\(\quad z_{\text{fwd}}=\mathrm{MLP}(z)\in\mathbb{R}^{B\times d}\) \hfill (uses Fwd tree)\\[0.5mm]
\textbf{Backward head (Spec \(\rightarrow\) Mol)}\\
\(\quad z_{\text{bwd}}=\mathrm{MLP}(z)\in\mathbb{R}^{B\times d}\) \hfill (uses Bwd tree)};

% Generator
\node[box, below right=12mm and 10mm of heads] (gen) {\textbf{GENERATOR} (SpecGen)\\
Input: \(z_{\text{fwd}}\)\\
MLP decoder\\
Output: synthetic spec\\
\(\hat{s}\in\mathbb{R}^{B\times n}\) (ReLU)};

\node[box, below left=12mm and 10mm of heads] (fwdReuse) {\textbf{Forward path reused}\\
Used for gen \& rerank\\
(Forward model provides\\
scores for candidates)};

% Adversarial loop + losses
\node[loss, right=25mm of gen] (loop) {\textbf{ADVERSARIAL TRAINING LOOP}\\
1) \textbf{Discriminator update:}\\
\(\;\;\bullet\) classify real vs fake\\
\(\;\;\bullet\) contrastive align: \(z_{\text{spec,real}}\approx z_{\text{fwd}}\)\\[0.5mm]
2) \textbf{Generator update:}\\
\(\;\;\bullet\) fool discriminator\\
\(\;\;\bullet\) spectrum recon: Wasserstein + cosine\\
\(\;\;\bullet\) cycle consistency (latent alignment)\\[1mm]
\(\mathcal{L}_{D}=\mathrm{BCE}(\text{real,fake})+\lambda\,\mathrm{InfoNCE}(z_{\text{spec}},z_{\text{fwd}})\)\\
\(\mathcal{L}_{G}=\mathrm{BCE}(\text{fake}\rightarrow\text{real})+\alpha_{w}\,W+\alpha_{c}\,C+\lambda_{\text{cycle}}\,\mathcal{L}_{\text{cycle}}\)};

\newcommand{\arrowVHV}[4][->]{%
  \draw[#1]
    (#2)
    -- ($(#2)+(0,-#4)$)
    -| ($(#3)+(0,#4)$)
    -- (#3);
}


% Connectors from input
\arrowVHV{input.south}{encMol.north}{6mm}
\arrowVHV{input.south}{encSpec.north}{6mm}
%\draw[arrow] (input.south) ++(-0.8,0) -| (encSpec.north);

% Encoder to discriminator & InfoNCE
\draw[arrow] (encSpec.south) -- (disc.north);
\draw[arrow] (disc.east) -- (infonce.west);

% Molecule encoder to fragment encoder / fusion
\draw[arrow] (encMol.south) -- (fragEnc.north);

% Fragment encoder to fusion
\draw[arrow] (fragEnc.south) -- (fusion.north);

% Latent fusion to heads
\draw[arrow] (fusion.south) -- (heads.north);

% Heads to generator and reuse box
\arrowVHV{heads.south}{gen.north}{6mm}
\arrowVHV{heads.south}{fwdReuse.north}{6mm}


% Generator output to discriminator as fake spectra (dashed)
\draw[dashedarrow] (gen.north east) ++(0.2,0) |- (disc.west);

% Loop box connections (conceptual)
\draw[arrow] (disc.south) -- (loop.north);
\draw[arrow] (gen.east) ++(0,-0.3) |- (loop.west);

% Frame around training block
\node[group, fit=(input) (encMol) (encSpec) (disc) (infonce) (fragEnc) (fusion) (heads) (gen) (fwdReuse) (loop),
      label={[title]above:TRAINING}] (trainGroup) {};

% =========================
% BOTTOM: INFERENCE & RETRIEVAL
% =========================
\node[title, below=16mm of trainGroup] (infTitle) {Inference \& Retrieval};

% Forward inference
\node[bigbox, below right=20mm and 6mm of infTitle, anchor=north west, xshift=-100mm] (infFwd) {\textbf{Forward (Mol \(\rightarrow\) Spec)}\\
Molecule \(\rightarrow\) EncMol \(\rightarrow z_{\text{mol}}\)\\
Fwd derivation tree \(\rightarrow\) FragSetEnc \(\rightarrow z_{\text{frag}}\)\\
\(z=(z_{\text{mol}}+z_{\text{frag}})/2\)\\
TaskHeads.fwd \(\rightarrow z_{\text{fwd}}\)\\
SpecGen \(\rightarrow \hat{s}\)};

% Backward inference
\node[bigbox, below left=20mm and 6mm of infTitle, anchor=north east, xshift=100mm] (infBwd) {\textbf{Backward (Spec \(\rightarrow\) Mol)}\\
Spectrum \(\rightarrow\) EncSpec \(\rightarrow z_{\text{spec}}\)\\
LatentIndex \(\rightarrow\) Top-\(K\) by cosine sim\\
Rerank with forward model\\
(use Bwd derivation tree)\\
Ranked SMILES candidates};

% Optional light connectors (decorative)
\draw[arrow] (infTitle.south) -- (infFwd.north);
\draw[arrow] (infTitle.south) -- (infBwd.north);

\node[group, fit=(infTitle) (infFwd) (infBwd),
      label={[title]above:INFERENCE}] (infGroup) {};

\end{tikzpicture}
