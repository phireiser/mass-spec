% ---- Shared math / text notation -----------------------------------------
% Add macros here that should mean the same thing in the thesis and the slides.

\providecommand{\msmap}{f : M \to S}    % forward map: structure -> spectrum
\providecommand{\msinv}{f^{-1} : S \to M} % inverse problem: spectrum -> structure
