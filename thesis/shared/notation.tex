% ---- Staged reveal inside a shared figure ---------------------------------
% A figure that is static in the report may be built up over overlays in the
% slides.  `visible on=<2->` / `dim on=<5>` on any node or path do that in
% beamer and are no-ops elsewhere, so one file serves both documents.
% Outside beamer \alt does not exist, hence the guard: the report simply gets
% the finished picture with every stage shown at once.
\makeatletter
\@ifundefined{alt}{%
  \tikzset{visible on/.style={}, dim on/.style={}}%
}{%
  \tikzset{
    invisible/.style={opacity=0, text opacity=0},
    dimmed/.style={opacity=0.16, text opacity=0.16},
    alt/.code args={<#1>#2#3}{\alt<#1>{\pgfkeysalso{#2}}{\pgfkeysalso{#3}}},
    visible on/.style={alt={#1{}{invisible}}},
    dim on/.style={alt={#1{dimmed}{}}},
  }%
}
\makeatother
