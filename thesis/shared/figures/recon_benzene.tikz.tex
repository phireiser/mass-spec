% Measured spectrum plotted upward, prediction mirrored downward.
\begin{tikzpicture}[
  fig,
  x=0.105cm, y=4.4cm,
  font=\sffamily\scriptsize,
  meas/.style={RoyalBlue, line width=1.3pt},
  pred/.style={BrickRed, line width=1.3pt},
  asg/.style={BrickRed!85!black, font=\sffamily\tiny},
]

% ---- axes -------------------------------------------------------------
\draw[fig line, black!55] (0,0) -- (92,0);
\node[anchor=west, font=\sffamily\scriptsize] at (92.5,0) {$m/z$};
\foreach \x in {0,20,40,60,80}{
  \draw[fig thin, black!55] (\x,0) -- (\x,-0.012);
}
% m/z ticks sit below the mirrored half so they never collide with the peaks
\foreach \x in {0,20,40,60,80}{
  \node[anchor=north, black!55, font=\sffamily\tiny] at (\x,-0.52) {\x};
}

% intensity scale (absolute value in both directions)
\foreach \y/\lab in {0.2/0.2, 0.4/0.4, 0.6/0.6}{

  \node[anchor=east, black!55, font=\sffamily\tiny] at (-1.8,\y) {\lab};

  \node[anchor=east, black!55, font=\sffamily\tiny] at (-1.8,-\y) {\lab};

% ---- measured spectrum (NIST), upward ---------------------------------

  {measured (NIST)};

% ---- predicted spectrum, mirrored downward ----------------------------
% every predicted peak with non-zero intensity is drawn, including the
% spurious m/z 1 at 0.010 -- nothing is filtered out of the prediction
\foreach \mz/\int in {1/0.0102, 26/0.1749, 51/0.1811, 77/0.2297, 78/0.4040}{

\node[BrickRed, anchor=west, font=\sffamily\scriptsize\bfseries] at (2,-0.455)
  {predicted (fragment-grounded decoder)};

% ---- fragment assignments, each hung off the left of its own peak so the
% ---- labels cannot run into one another (78 goes right, it is the deepest)
\node[asg, anchor=east] at (24.5,-0.170) {$26\;\mathrm{C_2H_2^+}$};
\node[asg, anchor=east] at (49.5,-0.181) {$51\;\mathrm{C_4H_3^+}$};

% cosine of this pair, as reported by the run

  {cosine $=0.758$};
