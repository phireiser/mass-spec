% ---------------------------------------------------------------------------
% One GINEConv layer: how a bond-aware message is built and applied.
% Left  : the 1-hop neighbourhood of atom v, with bond orders on the edges.
% Middle: per-neighbour message ReLU(h_u + e_uv), summed.
% Right : the update MLP applied to (1+eps) h_v + sum, plus this project's
%         LayerNorm/residual wrapper.
% Needs: tikz libraries arrows.meta, calc; \input{shared/notation} for `fig`.
% ---------------------------------------------------------------------------
\begin{tikzpicture}[
    fig,
    aC/.style={circle, draw=black!70, fill=black!11, minimum size=6.5mm,
               inner sep=0pt, font=\bfseries\footnotesize, text=black!70},
    aO/.style={circle, draw=BrickRed, fill=BrickRed!15, minimum size=6.5mm,
               inner sep=0pt, font=\bfseries\footnotesize, text=BrickRed},
    aV/.style={circle, draw=RoyalPurple, fill=RoyalPurple!18, minimum size=7.5mm,
               inner sep=0pt, font=\bfseries\footnotesize, text=RoyalPurple!70!black},
    hcell/.style={rectangle, minimum width=2.8mm, minimum height=4mm, inner sep=0pt},
    bondlab/.style={font=\tiny, fill=white, inner sep=1pt, text=NavyBlue},
    msgar/.style={-{Latex[length=1.6mm]}, semithick, RoyalPurple!80, dashed,
                  shorten >=1.5pt, shorten <=1.5pt},
    biga/.style={-{Latex[length=3.2mm, width=2.6mm]}, fig heavy, black!55},
    boxd/.style={rounded corners=2pt, draw=gray!55},
    figcap/.style={font=\scriptsize, align=center, text=black!75},
    ttl/.style={font=\bfseries\scriptsize, text=RoyalPurple},
]

% Feature strip: #1 name prefix, #2 x, #3 y, #4 colour, #5 label below
\newcommand{\hvec}[5]{%
  \begin{scope}[shift={(#2,#3)}]
    \foreach \k/\sh in {0/100, 1/62, 2/88, 3/45} {
      \node[hcell, fill=#4!\sh, draw=#4!\sh!black!45] at (0.3*\k, 0) {};
    }
    \coordinate (#1l) at (-0.22, 0);
    \coordinate (#1r) at ( 1.12, 0);
    \node[font=\tiny, anchor=west, text=black!70] at (1.20, 0) {#5};
  \end{scope}}

% =====================================================================
% PANEL 1 -- neighbourhood of v with bond orders
% =====================================================================
\node[ttl, anchor=south] at (0, 2.35) {1 -- neighbourhood of $v$};
\draw[boxd] (-2.0, -1.9) rectangle (2.0, 2.2);

\node[aV] (v)  at (0, 0)        {C};
\node[aC] (u1) at (-1.25, 1.15) {C};
\node[aC] (u2) at (-1.25,-1.15) {C};
\node[aO] (u3) at ( 1.30, 0.95) {O};

\draw[fig medium, black!75, dashed] (v) -- (u1)
      node[bondlab, midway, xshift=-2.0mm, yshift=-2.0mm] {$1.5$};
\draw[fig medium, black!75] (v) -- (u2)
      node[bondlab, midway, xshift=-1.0mm, yshift=-2.6mm] {$1.0$};
\draw[fig medium, black!75, double distance=1.1pt] (v) -- (u3)
      node[bondlab, midway, xshift=-1.6mm, yshift=2.2mm] {$2.0$};
\foreach \u in {u1, u2, u3} { \draw[msgar] (\u) to[bend left=26] (v); }

\node[figcap, anchor=north] at (0, -1.95)
    {atom features $h_u$ ($18$\,ch.)\\bond order $\rightarrow e_{uv}$};

% =====================================================================
% PANEL 2 -- per-neighbour messages, summed
% =====================================================================
\def\xB{6.1}
\node[ttl, anchor=south] at (\xB, 2.35) {2 -- bond-aware messages};
\draw[boxd] (\xB-2.5, -1.9) rectangle (\xB+2.6, 2.2);

\foreach \i/\yy/\col/\e in {1/1.35/gray/1.5, 2/0.35/gray/1.0, 3/-0.65/BrickRed/2.0} {
    \hvec{m\i}{\xB-2.20}{\yy}{\col}{}
    \node[font=\scriptsize] at (\xB-0.75, \yy) {$+$};
    \node[hcell, minimum width=6.5mm, fill=NavyBlue!18, draw=NavyBlue!60,
          font=\tiny, text=NavyBlue] at (\xB-0.30, \yy) {$\e$};
    \node[font=\scriptsize, text=black!70] at (\xB+0.45, \yy) {$\xrightarrow{\ \mathrm{ReLU}\ }$};
}
\node[circle, draw=RoyalPurple, thick, minimum size=7.5mm, inner sep=0pt,
      font=\small, text=RoyalPurple] (sig) at (\xB+1.85, 0.35) {$\Sigma$};
\foreach \yy in {1.35, 0.35, -0.65} { \draw[msgar] (\xB+0.95, \yy) -- (sig); }

\node[figcap, anchor=north] at (\xB, -1.95)
    {$h_u$ and $e_{uv}$ are both linearly lifted\\
     to $128$ channels, then \emph{added}};

% =====================================================================
% PANEL 3 -- the update
% =====================================================================
\def\xC{12.4}
\node[ttl, anchor=south] at (\xC, 2.35) {3 -- update of $h_v$};
\draw[boxd] (\xC-2.3, -1.9) rectangle (\xC+2.5, 2.2);

\hvec{hv}{\xC-1.95}{1.30}{RoyalPurple}{}
\node[font=\scriptsize, text=RoyalPurple!70!black, anchor=south]
      at (\xC-1.50, 1.58) {$(1{+}\epsilon)\,h_v$};
\node[font=\small] at (\xC-1.50, 0.78) {$+$};
\node[circle, draw=RoyalPurple, minimum size=6.5mm, inner sep=0pt,
      font=\scriptsize, text=RoyalPurple] at (\xC-1.50, 0.30) {$\Sigma$};
\node[font=\tiny, text=black!65, anchor=north] at (\xC-1.50, -0.10) {messages};

\begin{scope}[shift={(\xC+0.45, 0.75)}]
    \draw[boxd, draw=gray!70] (-0.62,-0.62) rectangle (0.62, 0.62);
    \foreach \n/\y in {a/0.35, b/0, c/-0.35} {
        \node[circle, draw=gray!80, fill=white, minimum size=2.2mm, inner sep=0pt] (li\n) at (-0.38,\y) {};}
    \foreach \n/\y in {a/0.22, b/-0.22} {
        \node[circle, draw=gray!80, fill=white, minimum size=2.2mm, inner sep=0pt] (lo\n) at (0.38,\y) {};}
    \foreach \a in {a,b,c} { \foreach \b in {a,b} { \draw[gray!60, thin] (li\a) -- (lo\b); } }
    \node[font=\scriptsize, anchor=north, text=black!70] at (0,-0.72) {MLP$_\theta$};
\end{scope}
\draw[biga] (\xC-0.95, 0.75) -- (\xC-0.25, 0.75);
\hvec{hn}{\xC+1.35}{0.75}{OliveGreen}{}
\node[font=\scriptsize, text=OliveGreen!45!black, anchor=north] at (\xC+1.80, 0.45) {$h_v'$};

\node[figcap, anchor=north] at (\xC, -1.95)
    {wrapped here as\\$h_v \!\leftarrow\! h_v + \mathrm{Drop}(\mathrm{ReLU}(\mathrm{LN}(h_v')))$};

\draw[biga] (2.15, 0.3) -- (\xB-2.65, 0.3);
\draw[biga] (\xB+2.75, 0.3) -- (\xC-2.45, 0.3);

% =====================================================================

% =====================================================================
\node[draw=RoyalPurple, thick, rounded corners=2pt, inner sep=5pt,
      font=\small, fill=RoyalPurple!4]
    at (6.2, -3.85)
    {$\displaystyle h_v^{(k+1)} = \mathrm{MLP}_\theta\!\Big((1+\epsilon)\,h_v^{(k)}
      \;+\!\! \sum_{u\in\mathcal{N}(v)}\!\! \mathrm{ReLU}\big(h_u^{(k)} + e_{uv}\big)\Big),
      \qquad \epsilon = 0 \ \text{(fixed)}$};
