% Pooling (graph readout): variable-size set of atom embeddings -> one vector.

% Needs: tikz libraries arrows.meta, calc; \input{shared/notation} for `fig`.
% ---------------------------------------------------------------------------
\begin{tikzpicture}[
    fig,
    aC/.style={circle, draw=black!70, fill=black!11, minimum size=5.5mm,
               inner sep=0pt, font=\bfseries\tiny, text=black!70},
    aO/.style={circle, draw=BrickRed, fill=BrickRed!15, minimum size=5.5mm,
               inner sep=0pt, font=\bfseries\tiny, text=BrickRed},
    aN/.style={circle, draw=RoyalBlue, fill=RoyalBlue!15, minimum size=5.5mm,

    num/.style={rectangle, minimum width=6.2mm, minimum height=4.4mm,

% one row of three numbers: #1 x (first cell centre), #2 y, #3 colour, #4..#6 values
\newcommand{\hrow}[6]{%
  \begin{scope}[shift={(#1,#2)}]
    \foreach \k/\val/\sh in {0/#4/100, 1/#5/62, 2/#6/88} {
      \node[num, fill=#3!\sh!white!35, draw=#3!55] at (0.66*\k, 0) {\val};
    }
  \end{scope}}

\def\ytop{3.35}   % panel top
\def\ybot{-3.35}  % panel bottom

% =====================================================================
% PANEL 1 -- n atom embeddings
% =====================================================================
\node[ttl, anchor=south] at (0, \ytop+0.12) {1 -- one vector per atom};
\draw[boxd] (-2.15, \ybot) rectangle (2.15, \ytop);

\foreach \i/\y/\col in {1/2.45/BrickRed, 2/1.65/black, 3/0.85/RoyalBlue, 4/0.05/black} {
  \foreach \k/\sh in {0/100, 1/60, 2/85, 3/45} {
    \node[rectangle, minimum width=3.4mm, minimum height=4.2mm, inner sep=0pt,
          fill=\col!\sh!white!30, draw=\col!30] at (-0.75+0.37*\k, \y) {};}
  \node[font=\tiny, text=black!65, anchor=west] at (0.65, \y) {$h_\i$};}

\node[figcap, anchor=north, text=black!70] at (0, -2.15)

% =====================================================================
% PANEL 2 -- the operators on a worked example
% =====================================================================
