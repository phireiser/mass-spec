\definecolor{Avec}{RGB}{210,55,40}
\definecolor{Bvec}{RGB}{40,120,200}
\definecolor{Wamber}{RGB}{180,120,20}
\definecolor{gray60}{RGB}{120,120,120}

\begin{tikzpicture}[x=1.35cm,y=1.35cm,>={Stealth[length=3mm]}]

  % coordinates
  \coordinate (O) at (0,0);
  \coordinate (A) at (6,1.5);     % vector A
  \coordinate (B) at (2.5,4);     % vector B
  \coordinate (F) at (3.294,0.8235); % foot of perpendicular from B onto line OA

  % faint reference axes
  \draw[gray60,line width=0.3pt,-{Stealth[length=2mm]}] (-0.3,0) -- (6.6,0);
  \draw[gray60,line width=0.3pt,-{Stealth[length=2mm]}] (0,-0.3) -- (0,4.5);

  % projection construction (perpendicular drop, drawn under the vectors)
  \draw[gray60,dashed,line width=0.6pt] (B) -- (F);
  % right-angle mark at foot F
  \begin{scope}[shift={(F)}]
    \draw[gray60,line width=0.5pt]
      ($(0,0)!0.16!(B)$) -- ($($(0,0)!0.16!(B)$)+($(0,0)!0.16!(O)$)$) -- ($(0,0)!0.16!(O)$);
  \end{scope}

  % vectors
  \draw[Avec,line width=1.8pt,->] (O) -- (A) node[above right=-1pt,Avec,font=\small] {$A$};
  \draw[Bvec,line width=1.8pt,->] (O) -- (B) node[above=1pt,Bvec,font=\small] {$B$};

  % projection segment, drawn on top of vector A so it stays visible
  \draw[Wamber,line width=2.4pt] (O) -- (F)
        node[midway,below=2pt,Wamber,font=\small] {$\|B\|\cos\theta$};

  % angle between them
  \pic[draw=Wamber,line width=1pt,angle radius=1.15cm,
       angle eccentricity=1.28,"$\theta$"{Wamber,font=\small}]
       {angle = A--O--B};

  % origin dot
  \fill (O) circle (1.6pt) node[below left=-1pt,font=\small] {$O$};

  % length labels along the vectors
  \node[Avec,font=\scriptsize] at ($(O)!0.55!(A)+(0,-0.28)$) {$\|A\|$};
  \node[Bvec,font=\scriptsize] at ($(O)!0.55!(B)+(-0.32,0.05)$) {$\|B\|$};

\end{tikzpicture}
