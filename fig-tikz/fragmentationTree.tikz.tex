% Small reusable molecular-graph glyph, centred on the given node.
\newcommand{\molGlyph}[1]{%
  \begin{scope}[shift={(#1)}, line width=0.3pt]
    \draw (0,-2.2) -- (0,0) -- (-2.2,1.3) (0,0) -- (2.2,1.3);
    \fill (0,0) circle[radius=0.55]
          (0,-2.2) circle[radius=0.55]
          (-2.2,1.3) circle[radius=0.55]
          (2.2,1.3) circle[radius=0.55];
  \end{scope}%
}

\newcommand{\fragmentationTree}{%
  \begin{tikzpicture}[
      x=0.5mm, y=-0.5mm,
      >=Latex,
      fragnode/.style={draw, ellipse, minimum width=6mm, minimum height=5mm,
                       inner sep=0pt, line width=0.4pt},
      treearrow/.style={->, line width=0.4pt, shorten >=0.5pt, shorten <=0.5pt},
    ]
    % ---- nodes (centres in original SVG coordinates) ----
    \node[fragnode] (root) at (44.5,  7.7) {};
    \node[fragnode] (A)    at (31.5, 29.2) {};
    \node[fragnode] (B)    at (57.7, 29.5) {};
    \node[fragnode] (A1)   at ( 8.6, 50.6) {};
    \node[fragnode] (A2)   at (26.5, 50.5) {};
    \node[fragnode] (A3)   at (43.9, 50.6) {};
    \node[fragnode] (B1)   at (61.5, 50.7) {};
    \node[fragnode] (A31)  at (28.7, 73.2) {};
    \node[fragnode] (A32)  at (45.1, 72.2) {};
    \node[fragnode] (B11)  at (63.5, 72.3) {};

    % ---- directed edges (parent -> child) ----
    \draw[treearrow] (root) -- (A);
    \draw[treearrow] (root) -- (B);
    \draw[treearrow] (A)    -- (A1);
    \draw[treearrow] (A)    -- (A2);
    \draw[treearrow] (A)    -- (A3);
    \draw[treearrow] (A3)   -- (A31);
    \draw[treearrow] (A3)   -- (A32);
    \draw[treearrow] (B)    -- (B1);
    \draw[treearrow] (B1)   -- (B11);

    % ---- fragment glyph inside every node ----
    \foreach \n in {root,A,B,A1,A2,A3,B1,A31,A32,B11}{\molGlyph{\n}}
  \end{tikzpicture}%
}
