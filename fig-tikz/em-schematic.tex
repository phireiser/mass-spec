\begin{tikzpicture}[x=1cm,y=1cm,line cap=round,line join=round,font=\sffamily]
\definecolor{msred}{RGB}{170,0,60}

% --------------------------
% Parameters (tuned for Fig 2.17)
% --------------------------
\def\xL{0.85}
\def\dxDyn{1.55}

\def\yTop{3.55} \def\yMid{2.20} \def\yLow{0.85}
% Right column is shifted by half a left-side gap so it sits between left dynodes.
\def\yTopR{2.875} \def\yMidR{1.525} \def\yLowR{0.175}

\def\rDyn{0.72}
\def\anodeY{0.05}

% centers
\coordinate (L1) at (\xL,\yTop);
\coordinate (L2) at (\xL,\yMid);
\coordinate (L3) at (\xL,\yLow);
\coordinate (R1) at ($(L1)+(\dxDyn,\yTopR-\yTop)$);
\coordinate (R2) at ($(L2)+(\dxDyn,\yMidR-\yMid)$);
\coordinate (R3) at ($(L3)+(\dxDyn,\yLowR-\yLow)$);



% --------------------------
% Dynodes
% --------------------------
\draw[msred,line width=1.10pt] ($(L1)+(0,\rDyn)$) arc[start angle=150,end angle=210,radius=\rDyn];
\draw[msred,line width=1.10pt] ($(L2)+(0,\rDyn)$) arc[start angle=150,end angle=210,radius=\rDyn];
\draw[msred,line width=1.10pt] ($(L3)+(0,\rDyn)$) arc[start angle=150,end angle=210,radius=\rDyn];
\draw[msred,line width=1.10pt] ($(R1)+(0,\rDyn)$) arc[start angle=30,end angle=-30,radius=\rDyn];
\draw[msred,line width=1.10pt] ($(R2)+(0,\rDyn)$) arc[start angle=30,end angle=-30,radius=\rDyn];

% --------------------------
% Anode
% --------------------------
\coordinate (A1) at ($(L3)+(-0.15,\anodeY-\yLow)$);
\coordinate (A2) at ($(R3)+( 0.75,\anodeY-\yLowR)$);
\coordinate (Astem) at ($(A1)!0.50!(A2)$);

\draw[msred,line width=1.15pt] (A1) -- (A2);
\draw[msred,line width=1.15pt] (Astem) -- ++(0,-0.95);

% --------------------------
% Electron cascade
% --------------------------
% --------------------------
% Electron cascade (multiplication at dynodes, like Fig. 2.17)
% --------------------------
\begin{scope}[on background layer]

  % --- impact points placed RELATIVE to dynode centers ---
  % (fractions tuned so paths sit between dynode arcs like in the PDF)
  \coordinate (P1) at ($(L1)!0.05!(R1) +(0,0.4)$); % on/near left-top dynode (inner face)
  \coordinate (P2) at ($(L1)!0.95!(R1)+ (0,0.3)$); % on/near right-top dynode (inner face)
  \coordinate (P3) at ($(L2)!0.05!(R2)+ (0,0.4)$); % on/near left-mid dynode
  \coordinate (P4) at ($(L2)!0.95!(R2)+ (0,0.3)$); % on/near right-mid dynode
  \coordinate (P5) at ($(L3)!0.05!(R2)+ (0,0.3)$); % bottom-left dynode (approached from right-mid)

   % Reusable fan-out stage: #1 source point, #2 target point, #3 number of incoming rays.
   \newcommand{\DrawCascadeStage}[3]{%
      \def\cnt{#3}%
      \pgfmathtruncatemacro{\kmax}{\cnt-1}%
      \foreach \k in {0,...,\kmax}{%
         \pgfmathsetmacro{\yoff}{2*\k/\kmax - 1}%
         \pgfmathsetmacro{\dyminus}{-1 + 2*(2*\k)/(2*\cnt - 1)}%
         \pgfmathsetmacro{\dyplus}{-1 + 2*(2*\k + 1)/(2*\cnt - 1)}%
         \draw[msred,line width=0.25pt]
             ($(#1)+(0,0.2*\yoff)$) -- ($(#2)+(0,0.2*\dyminus)$);
         \draw[msred,line width=0.25pt]
             ($(#1)+(0,0.2*\yoff)$) -- ($(#2)+(0,0.2*\dyplus)$);
      }%
   }

  % incoming ion (single)
  \coordinate (IonIn) at ($(P1)+(1.2,0.50)$);
  \draw[msred,line width=0.25pt] (IonIn) -- (P1);

  % --------------------------
  % Stage 1: first reflections
  % --------------------------
  \def\cnt{1}
  \pgfmathtruncatemacro{\kmax}{\cnt-1}
  \foreach \k in {0,...,\kmax} {
    \pgfmathsetmacro{\dy}{(\k / 1 - 1.0/ \cnt)}
    \pgfmathsetmacro{\yoff}{(\k / 1 - 0.0/ \cnt)}
    \draw[msred,line width=0.25pt]
      ($(P1)+(0,0.2*\yoff)$) -- ($(P2)+(0,0.2*\dy)$);
    \draw[msred,line width=0.25pt]
      ($(P1)+(0,0.2*\yoff)$) -- ($(P2)+(0,-0.2*\dy)$);
  }

  \DrawCascadeStage{P2}{P3}{2}
  \DrawCascadeStage{P3}{P4}{4}
  \DrawCascadeStage{P4}{P5}{8}

  \foreach \t in {0.03125,0.0625,...,0.98}{
    \draw[msred,line width=0.25pt] 
      ($(P5) + (0,0.2*\t)$) -- ($(A1)!\t!(A2)$);
  }

\end{scope}


% --------------------------
% Labels
% --------------------------
\node[anchor=south] at ($(R1)+(0.05,1.20)$) {Ion};

\node[anchor=east] at ($(L2)+(-1.05,0.05)$) {Dynodes};
\draw[->,black,line width=0.9pt] ($(L2)+(-0.90,0.05)$) -- ($(L1)+(-0.2,0.1)$);
\draw[->,black,line width=0.9pt] ($(L2)+(-0.90,0.05)$) -- ($(L3)+(-0.2,0.6)$);
\draw[->,black,line width=0.9pt] ($(L2)+(-0.90,0.05)$) -- ($(L2)+(-0.2,0.1)$);

\node[anchor=west,align=left] at ($(R2)+(1.05,-0.15)$) {Electron\\[-10pt]cascade};
%\draw[black,line width=0.9pt] ($(R2)+(0.90,-0.25)$) -- ($(S0)+(-0.05,-0.15)$);

\node[anchor=east] at ($(A1)+(-0.10,-0.40)$) {Anode};
\draw[->,black,line width=0.9pt] ($(A1)+(0.20,-0.28)$) -- ($(Astem)+(-0.10,-0.1)$);

\end{tikzpicture}
